function getTable(name: string): Map<string, any> {
  if (!h.tables[name]) h.tables[name] = new Map();
  return h.tables[name];
}

function matchRow(row: any, where: any): boolean {
  for (const [k, v] of Object.entries(where || {})) {
    if (k.includes("_") && v !== null && typeof v === "object" && !(v instanceof Date) && !Array.isArray(v)) {
      for (const [k2, v2] of Object.entries(v)) {
        if (row[k2] !== v2) return false;
      }
      continue;
    }
    if (v !== null && typeof v === "object" && !(v instanceof Date) && !Array.isArray(v)) {
      for (const [op, ov] of Object.entries(v)) {
        const rv = row[k];
        if (op === "not") {
          if (rv === ov) return false;
        } else if (op === "gt") {
          if (!(rv > ov)) return false;
        } else if (op === "lt") {
          if (!(rv < ov)) return false;
        } else if (op === "in") {
          if (!Array.isArray(ov) || !ov.includes(rv)) return false;
        } else {
          return false; // 未实现操作符显式失败
        }
        continue;
      }
      continue;
    }
    const rv = row[k];
    if (v instanceof Date || rv instanceof Date) {
      const a = rv instanceof Date ? rv.getTime() : rv;
      const b = v instanceof Date ? v.getTime() : v;
      if (a !== b) return false;
    } else if (rv !== v) return false;
  }
  return true;
}

function applyData(row: any, data: any): any {
  const next = { ...row };
  for (const [k, v] of Object.entries(data || {})) {
    if (v !== null && typeof v === "object" && !(v instanceof Date) && !Array.isArray(v) && "increment" in v) {
      next[k] = Number(next[k] ?? 0) + Number((v as any).increment);
    } else {
      next[k] = v;
    }
  }
  return next;
}

function selectFields(row: any, select: any) {
  if (!row) return null;
  if (!select) return { ...row };
  const out: any = {};
  for (const k of Object.keys(select)) if (select[k]) out[k] = row[k];
  return out;
}

function makePrisma() {
  let txDepth = 0;
  let txQueue: Promise<any> = Promise.resolve();
  let currentUndo: (() => void)[] | null = null;
  const record = (kind: string, model: string, method: string) => h.writes.push(`${kind} ${model}.${method}`);

  const makeClient = (kind: "ROOT" | "TX") => {
    const model = (name: string) => {
      const rows = () => [...getTable(name).values()];
      const findRow = (where: any) => rows().find((r) => matchRow(r, where)) ?? null;
      const doWrite = (method: string, mutate: () => any): any => {
        record(kind, name, method);
        if (kind === "ROOT" && txDepth > 0) throw new Error("ROOT_WRITE_DURING_TX");
        // [R0-H1 更正] 事务内写前采集快照（旧夹具写后采集，无法证明回滚）
        const undo = kind === "TX" ? currentUndo : null;
        const snapshotRows = undo ? new Map(getTable(name)) : null;
        const result = mutate();
        if (undo && snapshotRows) {
          const snap = snapshotRows;
          undo.push(() => {
            h.tables[name] = snap;
          });
        }
        return result;
      };
      return {
        findUnique: async ({ where, select }: any) => selectFields(findRow(where) ?? (null as any), select),
        findFirst: async ({ where, select }: any) => selectFields(findRow(where) ?? (null as any), select),
        findMany: async ({ where, select }: any) =>
          rows()
            .filter((r) => matchRow(r, where))
            .map((r) => selectFields(r, select)),
        count: async ({ where }: any) => rows().filter((r) => matchRow(r, where)).length,
        create: async ({ data }: any) =>
          doWrite("create", () => {
            const row = { ...data };
            if (!row.id) row.id = `auto_${Math.random().toString(36).slice(2)}`;
            getTable(name).set(row.id, row);
            return { ...row };
          }),
        update: async ({ where, data }: any) =>
          doWrite("update", () => {
            const row = findRow(where);
            if (!row) throw new Error(`P2025:${name}`);
            const next = applyData(row, data);
            getTable(name).set(row.id ?? where.id, next);
            return { ...next };
          }),
        updateMany: async ({ where, data }: any) =>
          doWrite("updateMany", () => {
            let count = 0;
            for (const row of rows()) {
              if (matchRow(row, where)) {
                getTable(name).set(row.id, applyData(row, data));
                count++;
              }
            }
            return { count };
          }),
        upsert: async ({ where, create, update }: any) =>
          doWrite("upsert", () => {
            const row = findRow(where);
            if (row) {
              const next = applyData(row, update);
              getTable(name).set(row.id, next);
              return { ...next };
            }
            const created = { ...create };
            if (!created.id) created.id = `auto_${Math.random().toString(36).slice(2)}`;
            getTable(name).set(created.id, created);
            return { ...created };
          }),
      };
    };
    return new Proxy({}, { get: (_t, prop: string) => model(prop) });
  };

  const prismaProxy: any = new Proxy(
    {},
    {
      get: (_t, prop: string) => {
        if (prop === "$transaction") {
          return (fn: any) => {
            const run = async () => {
              txDepth++;
              const undo: (() => void)[] = [];
              const saved = currentUndo;
              currentUndo = undo;
              try {
                const r = await fn(makeClient("TX"));
                if (h.fault.commit) {
                  h.fault.commit = false;
                  throw new Error("INJECTED_COMMIT_FAIL");
                }
                return r;
              } catch (e) {
                for (let i = undo.length - 1; i >= 0; i--) undo[i]();
                throw e;
              } finally {
                currentUndo = saved;
                txDepth--;
              }
            };
            const p = txQueue.then(run, run);
            txQueue = p.catch(() => {});
            return p;
          };
        }
        return (makeClient("ROOT") as any)[prop];
      },
    },
  );
  return prismaProxy;
}

describe("R0-H1 纯准入函数：可信边界与金额（M01/M02/M04）", async () => {
  const { decidePlatformModelCostAdmission: decide, PROTOCOL_FRAMING_TOKEN_ALLOWANCE: FRAMING } = await import(
    "@/lib/model-cost-admission"
  );
  const { estimateCombinedTokenCount } = await import("@/lib/token-estimate");

  // 独立复算（不调用被测函数）：输入上界 = UTF-8 字节数 + 协议封装余量（与文档化规则一致）
  function independentCostPoints(
    systemPrompt: string,
    userPrompt: string,
    maxOutputTokens: number,
    costIn: number,
    costOut: number,
    markupBps: number,
  ): { bound: number; points: number } {
    const bound = new TextEncoder().encode(systemPrompt).length + new TextEncoder().encode(userPrompt).length + 128;
    const costMicros = (BigInt(bound) * BigInt(costIn) + BigInt(maxOutputTokens) * BigInt(costOut) + BigInt(999999)) / BigInt(1000000);
    const saleMicros = (costMicros * BigInt(10000 + markupBps) + BigInt(9999)) / BigInt(10000);
    const points = (saleMicros + BigInt(9999)) / BigInt(10000);
    return { bound, points: Number(points) };
  }

  it("M04：100 点预算的真实 100/101/150/200/201 边界（独立复算；修正旧用例参数错误）", () => {
    // 输入上界精确 = 1,000,000 Token：systemPrompt 999,872 字节（ASCII）+ userPrompt 0 + 封装 128
    const systemPrompt = "x".repeat(1_000_000 - 128);
    const userPrompt = "";
    // 1e6 Token × costIn 微元/百万 = costIn 微元 → 点数 = costIn / 10000
    const cases: [number, boolean][] = [
      [1_000_000, true], // 恰好 100 点 → 放行（等值）
      [1_010_000, false], // 101 点
      [1_500_000, false], // 150 点
      [2_000_000, false], // 200 点
      [2_010_000, false], // 201 点
    ];
    for (const [costIn, expectAllowed] of cases) {
      const independent = independentCostPoints(systemPrompt, userPrompt, 0, costIn, 0, 0);
      const expectedPoints = costIn / 10_000;
      expect(independent.points, `case ${costIn}`).toBe(expectedPoints); // 独立复算吻合
      const d = decide({
        pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: costIn, costOutputMicrosPerMillion: 0 },
        systemPrompt,
        userPrompt,
        maxOutputTokens: 0,
        budgetPoints: 100,
        markupRateBps: 0,
      });
      expect(d.allowed, `case ${costIn}`).toBe(expectAllowed);
      if (d.allowed) expect(d.inputTokenUpperBound).toBe(1_000_000);
    }
  });

  it("M01：字节上界被采用——字符粗估能过而字节上界不过的请求被拒绝（独立对照）", () => {
    // 1000 个非 CJK 字符（每个 2 UTF-8 字节）：字符粗估约数百 Token，字节上界约 2100+
    const cjkFree = "Ж".repeat(1000);
    const charEstimate = estimateCombinedTokenCount("提示词", cjkFree);
    const bound = new TextEncoder().encode(cjkFree).length + 128;
    expect(bound).toBeGreaterThan(charEstimate * 2); // 两者显著分离
    const costIn = 500_000_000; // 字节上界折算 > 100 点；字符粗估折算 < 100 点
    const viaChar = costPointsIf(charEstimate);
    const viaBound = costPointsIf(bound);
    expect(viaChar).toBeLessThan(100); // 旧口径（字符粗估）会放行
    expect(viaBound).toBeGreaterThan(100); // 新口径（字节上界）必须拒绝
    function costPointsIf(tokens: number): number {
      return Number((BigInt(tokens) * BigInt(costIn) + BigInt(999999)) / BigInt(1000000) / BigInt(10000)) + 1;
    }
    const d = decide({
      pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: costIn, costOutputMicrosPerMillion: 0 },
      systemPrompt: "提示词",
      userPrompt: cjkFree,
      maxOutputTokens: 0,
      budgetPoints: 100,
      markupRateBps: 0,
    });
    expect(d.allowed).toBe(false);
    if (!d.allowed) expect(d.code).toBe("MODEL_COST_EXCEEDS_BUDGET");
  });

  it("M01：正常短材料 + 低单价 → 可靠边界下正常放行（不永久拒绝）", () => {
    const d = decide({
      pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 1000, costOutputMicrosPerMillion: 1000 },
      systemPrompt: "你是一名专业分析师。",
      userPrompt: "这是一段正常的文本输入材料。",
      maxOutputTokens: 300,
      budgetPoints: 100,
      markupRateBps: 0,
    });
    expect(d.allowed).toBe(true);
    if (d.allowed) expect(d.inputBoundSource).toBe("UTF8_BYTES_PLUS_FRAMING");
  });

  it("M02：未知费用规则 / 非零缓存类别 → 拒绝；空规则与明确免费对照放行", () => {
    const base = { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 1000, costOutputMicrosPerMillion: 1000 };
    // 未知规则拒绝
    const r1 = decide({ pricing: { ...base, rules: { unknownSurcharge: { rate: "5%" } } }, systemPrompt: "s", userPrompt: "u", maxOutputTokens: 100, budgetPoints: 100, markupRateBps: 0 });
    expect(r1.allowed).toBe(false);
    if (!r1.allowed) expect(r1.code).toBe("MODEL_COST_NOT_TRUSTWORTHY");
    // 非零缓存成本拒绝（本次是否命中缓存不可预知）
    const r2 = decide({ pricing: { ...base, costCacheReadMicrosPerMillion: 500 }, systemPrompt: "s", userPrompt: "u", maxOutputTokens: 100, budgetPoints: 100, markupRateBps: 0 });
    expect(r2.allowed).toBe(false);
    // 空规则对象 / null rules → 不误拦
    for (const rules of [null, {}]) {
      const d = decide({ pricing: { ...base, rules }, systemPrompt: "s", userPrompt: "u", maxOutputTokens: 100, budgetPoints: 100, markupRateBps: 0 });
      expect(d.allowed).toBe(true);
    }
    // 明确免费对照（FREE 与 VERIFIED+0/0）
    for (const pricing of [
      { currency: "CNY", priceStatus: "FREE", costInputMicrosPerMillion: 0, costOutputMicrosPerMillion: 0, rules: null },

      expect(d.allowed).toBe(true);
      if (d.allowed) expect(d.conservativeCostPoints).toBe(0);
    }
    // 基础价格校验保持：缺失/不可信状态/币种/单侧缺失仍拒绝
    for (const pricing of [null, { ...base, priceStatus: "OBSERVED_ONLY" }, { ...base, currency: "USD" }, { ...base, costInputMicrosPerMillion: null }, { ...base, costOutputMicrosPerMillion: null }]) {
      const d = decide({ pricing, systemPrompt: "s", userPrompt: "u", maxOutputTokens: 100, budgetPoints: 100, markupRateBps: 0 });
      expect(d.allowed).toBe(false);
      if (!d.allowed) expect(d.code).toBe("MODEL_COST_NOT_TRUSTWORTHY");
