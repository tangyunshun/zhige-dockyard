function getTable(name: string): Map<string, any> {
  if (!h.tables[name]) h.tables[name] = new Map();
  return h.tables[name];
}

function matchRow(row: any, where: any): boolean {
  for (const [k, v] of Object.entries(where || {})) {
    // 复合键（userId_workspaceId 等）→ AND 子字段
    if (k.includes("_") && v !== null && typeof v === "object" && !(v instanceof Date) && !Array.isArray(v)) {
      for (const [k2, v2] of Object.entries(v)) {
        if (row[k2] !== v2) return false;
      }
      continue;
    }
    if (v !== null && typeof v === "object" && !(v instanceof Date) && !Array.isArray(v)) {
      for (const [op, ov] of Object.entries(v)) {
        const rv = row[k];
        if (op === "not") {
          if (rv === ov) return false;
        } else if (op === "gt") {
          if (!(rv > ov)) return false;
        } else if (op === "lt") {
          if (!(rv < ov)) return false;
        } else if (op === "in") {
          if (!Array.isArray(ov) || !ov.includes(rv)) return false;
        } else if (op === "increment") {
          // 递增不是条件
        } else {
          return false; // 未实现操作符显式失败
        }
        continue;
      }
      continue;
    }
    const rv = row[k];
    if (v instanceof Date || rv instanceof Date) {
      const a = rv instanceof Date ? rv.getTime() : rv;
      const b = v instanceof Date ? v.getTime() : v;
      if (a !== b) return false;
    } else if (rv !== v) return false;
  }
  return true;
}

function applyData(row: any, data: any): any {
  const next = { ...row };
  for (const [k, v] of Object.entries(data || {})) {
    if (v !== null && typeof v === "object" && !(v instanceof Date) && !Array.isArray(v) && "increment" in v) {
      next[k] = Number(next[k] ?? 0) + Number((v as any).increment);
    } else {
      next[k] = v;
    }
  }
  return next;
}

function selectFields(row: any, select: any) {
  if (!row) return null;
  if (!select) return { ...row };
  const out: any = {};
  for (const k of Object.keys(select)) if (select[k]) out[k] = row[k];
  return out;
}

function makePrisma() {
  let txDepth = 0;
  let txQueue: Promise<any> = Promise.resolve();
  const record = (kind: string, model: string, method: string) => h.writes.push(`${kind} ${model}.${method}`);

  const makeClient = (kind: "ROOT" | "TX") => {
    const model = (name: string) => {
      const rows = () => [...getTable(name).values()];
      const findRow = (where: any) => rows().find((r) => matchRow(r, where)) ?? null;
      const doWrite = (method: string, run: () => any, undoStore: (() => void)[] | null) => {
        record(kind, name, method);
        if (kind === "ROOT" && txDepth > 0) throw new Error("ROOT_WRITE_DURING_TX");
        const result = run();
        if (undoStore) {
          const snapshotRows = new Map(getTable(name));
          const afterRows = new Map(getTable(name));
          undoStore.push(() => {
            h.tables[name] = snapshotRows;
            void afterRows;
          });
        }
        return result;
      };
      return {
        findUnique: async ({ where, select }: any) => selectFields(findRow(where) ?? (null as any), select),
        findFirst: async ({ where, select }: any) => selectFields(findRow(where) ?? (null as any), select),
        findMany: async ({ where, select }: any) =>
          rows()
            .filter((r) => matchRow(r, where))
            .map((r) => selectFields(r, select)),
        count: async ({ where }: any) => rows().filter((r) => matchRow(r, where)).length,
        create: async ({ data }: any) =>
          doWrite("create", () => {
            const row = { ...data };
            if (!row.id) row.id = `auto_${Math.random().toString(36).slice(2)}`;
            getTable(name).set(row.id, row);
            return { ...row };
          }, kind === "TX" ? currentUndo : null),
        update: async ({ where, data }: any) =>
          doWrite("update", () => {
            const row = findRow(where);
            if (!row) throw new Error(`P2025:${name}`);
            const next = applyData(row, data);
            getTable(name).set(row.id ?? where.id, next);
            return { ...next };
          }, kind === "TX" ? currentUndo : null),
        updateMany: async ({ where, data }: any) =>
          doWrite("updateMany", () => {
            let count = 0;
            for (const row of rows()) {
              if (matchRow(row, where)) {
                getTable(name).set(row.id, applyData(row, data));
                count++;
              }
            }
            return { count };
          }, kind === "TX" ? currentUndo : null),
        upsert: async ({ where, create, update }: any) =>
          doWrite("upsert", () => {
            const row = findRow(where);
            if (row) {
              const next = applyData(row, update);
              getTable(name).set(row.id, next);
              return { ...next };
            }
            const created = { ...create };
            if (!created.id) created.id = `auto_${Math.random().toString(36).slice(2)}`;
            getTable(name).set(created.id, created);
            return { ...created };
          }, kind === "TX" ? currentUndo : null),
      };
    };
    return new Proxy(
      {},
      { get: (_t, prop: string) => model(prop) },
    );
  };

  let currentUndo: (() => void)[] | null = null;
  const prismaProxy: any = new Proxy(
    {},
    {
      get: (_t, prop: string) => {
        if (prop === "$transaction") {
          return (fn: any) => {
            const run = async () => {
              txDepth++;
              const undo: (() => void)[] = [];
              const saved = currentUndo;
              currentUndo = undo;
              try {
                const r = await fn(makeClient("TX"));
                if (h.fault.commit) {
                  h.fault.commit = false;
                  throw new Error("INJECTED_COMMIT_FAIL");
                }
                return r;
              } catch (e) {
                for (let i = undo.length - 1; i >= 0; i--) undo[i]();
                throw e;
              } finally {
                currentUndo = saved;
                txDepth--;
              }
            };
            const p = txQueue.then(run, run);
            txQueue = p.catch(() => {});
            return p;
          };
        }
        return (makeClient("ROOT") as any)[prop];
      },
    },
  );
  return prismaProxy;
}

// ══════════════ 纯准入函数（L01/L02/L04） ══════════════

describe("R0-H 纯准入函数 decidePlatformModelCostAdmission", async () => {
  const { decidePlatformModelCostAdmission: decide } = await import("@/lib/model-cost-admission");
  const bigInput = 50_000; // 5 万 Token 输入
  const outCap = 3000;

  function costPointsIndependent(inputTokens: number, outputTokens: number, costIn: number, costOut: number, markupBps: number): number {
    // 独立复算（BigInt 微元，向上取整），不调用被测函数
    const costMicros = (BigInt(inputTokens) * BigInt(costIn) + BigInt(outputTokens) * BigInt(costOut) + BigInt(999999)) / BigInt(1000000);
    const saleMicros = (costMicros * BigInt(10000 + markupBps) + BigInt(9999)) / BigInt(10000);
    const points = (saleMicros + BigInt(9999)) / BigInt(10000);
    return Number(points);
  }

  it("L01：100 点预算下，101/150/200/201 点成本均拒绝；等于/低于预算放行", () => {
    // 5 万输入 + 3000 输出，输入单价 2,000,000 微元/百万（=2 元/百万 → 5 万 Token=0.1 元=10 点基价）… 直接按单价换算
    // 50,000 输入 Token = 0.05 百万；costIn 微元/百万 → 点数 = ceil(0.05 × costIn 微元 / 10000)
    // 101 点 → costIn = 202,000,000；150 → 300,000,000；200 → 400,000,000；201 → 402,000,000
    const cases: [number, boolean][] = [
      [20_200_000, false],
      [30_000_000, false],
      [40_000_000, false],
      [40_200_000, false],
    ];
    for (const [costIn, expectReject] of cases) {
      const d = decide({
        pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: costIn, costOutputMicrosPerMillion: 0 },
        inputTokens: bigInput,
        maxOutputTokens: outCap,
        budgetPoints: 100,
        markupRateBps: 0,
      });
      const independent = costPointsIndependent(bigInput, 0, costIn, 0, 0);
      expect(independent, `case ${costIn} 复算`).toBeGreaterThan(100); // 复算确认确实超预算
      expect(d.allowed, `case ${costIn} 准入`).toBe(false);
      if (!d.allowed) expect(d.code).toBe("MODEL_COST_EXCEEDS_BUDGET");
    }
    // 等于/低于预算（80 点）放行
    const dOk = decide({
      pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 16_000_000, costOutputMicrosPerMillion: 0 },
      inputTokens: bigInput,
      maxOutputTokens: outCap,
      budgetPoints: 100,
      markupRateBps: 0,
    });
    const independent = costPointsIndependent(bigInput, 0, 16_000_000, 0, 0);
    expect(independent).toBe(80);
    expect(dOk.allowed).toBe(true);
    if (dOk.allowed) expect(dOk.conservativeCostPoints).toBe(80);
  });

  it("L01：成本恰好等于预算 → 放行（无两倍容忍区，也不得低于预算拒绝）", () => {
    const d = decide({
      pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 2_000_000, costOutputMicrosPerMillion: 0 },
      inputTokens: 50_000,
      maxOutputTokens: 0,
      budgetPoints: 100,
      markupRateBps: 0,
    });
    expect(d.allowed).toBe(true); // 精确 100 点 = 预算
  });

  it("L02：pricing 缺失 / 状态不可信 / 币种错误 / 单侧缺失 / 非法值 → 全部 MODEL_COST_NOT_TRUSTWORTHY", () => {
    const base = { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 1000, costOutputMicrosPerMillion: 1000 };
    const badCases: [string, any][] = [
      ["pricing 缺失", null],
      ["状态 UNCONFIGURED", { ...base, priceStatus: "UNCONFIGURED" }],
      ["状态 UNVERIFIED", { ...base, priceStatus: "UNVERIFIED" }],
      ["状态 OBSERVED_ONLY", { ...base, priceStatus: "OBSERVED_ONLY" }],
      ["币种 USD", { ...base, currency: "USD" }],
      ["输入侧缺失", { ...base, costInputMicrosPerMillion: null }],
      ["输出侧缺失", { ...base, costOutputMicrosPerMillion: null }],
      ["负数成本", { ...base, costInputMicrosPerMillion: -5 }],
      ["小数成本", { ...base, costInputMicrosPerMillion: 1.5 }],
    ];
    for (const [label, pricing] of badCases) {
      const d = decide({ pricing, inputTokens: 1000, maxOutputTokens: 100, budgetPoints: 100, markupRateBps: 0 });
      expect(d.allowed, label).toBe(false);
      if (!d.allowed) expect(d.code, label).toBe("MODEL_COST_NOT_TRUSTWORTHY");
    }
  });

  it("L02：明确免费（priceStatus=FREE 或 VERIFIED 且两侧 0）→ 零成本放行", () => {
    for (const pricing of [
      { currency: "CNY", priceStatus: "FREE", costInputMicrosPerMillion: 0, costOutputMicrosPerMillion: 0 },
      { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 0, costOutputMicrosPerMillion: 0 },
    ]) {
      const d = decide({ pricing, inputTokens: 100000, maxOutputTokens: 5000, budgetPoints: 100, markupRateBps: 0 });
      expect(d.allowed).toBe(true);
      if (d.allowed) expect(d.conservativeCostPoints).toBe(0);
    }
  });

  it("L04：单位/舍入/大值独立复算（BigInt 微元，向上取整；加价系数沿用既有口径）", () => {
    // 100 万输入 × 1,000,000 微元/百万 = 1 元 = 100 点；加价 5000bps（k=1.5，既有口径 10000+bps）→ 150 点
    const d = decide({
      pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 1_000_000, costOutputMicrosPerMillion: 0 },
      inputTokens: 1_000_000,
      maxOutputTokens: 0,
      budgetPoints: 150,
      markupRateBps: 5000,
    });
    const independent = costPointsIndependent(1_000_000, 0, 1_000_000, 0, 5000);
    expect(independent).toBe(150);
    expect(d.allowed).toBe(true);
    if (d.allowed) expect(d.conservativeCostPoints).toBe(150);
    // 预算 149 → 拒绝（>预算，哪怕只差 1 点）
    const d2 = decide({
      pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 1_000_000, costOutputMicrosPerMillion: 0 },
      inputTokens: 1_000_000,
      maxOutputTokens: 0,
      budgetPoints: 149,
      markupRateBps: 5000,
    });
    expect(d2.allowed).toBe(false);
    // 大值：10 亿 Token × 10 亿微元/百万（>2^53，BigInt 不溢出）
    const d3 = decide({
      pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 1_000_000_000, costOutputMicrosPerMillion: 0 },
      inputTokens: 1_000_000_000,
      maxOutputTokens: 0,
      budgetPoints: 1_000_000_000_000,
      markupRateBps: 0,
    });
    expect(d3.allowed).toBe(true);
    if (d3.allowed) expect(d3.conservativeCostPoints).toBe(100_000_000);
    // 损坏预算（小数点数会被 consumePoints 取整产生歧义）→ 拒绝
    const d4 = decide({
      pricing: { currency: "CNY", priceStatus: "VERIFIED", costInputMicrosPerMillion: 1000, costOutputMicrosPerMillion: 1000 },
      inputTokens: 1000,
      maxOutputTokens: 100,
      budgetPoints: 100.5,
      markupRateBps: 0,
    });
    expect(d4.allowed).toBe(false);
    if (!d4.allowed) expect(d4.code).toBe("MODEL_COST_NOT_TRUSTWORTHY");
  });
});

  it("超预算（成本 150 点 vs 预算 100 点）→ 400 EXECUTION_COST_EXCEEDS_BUDGET，零扣点/零流水/零 execute/无任务", async () => {

    expect(typeof r.body.error).toBe("string");
