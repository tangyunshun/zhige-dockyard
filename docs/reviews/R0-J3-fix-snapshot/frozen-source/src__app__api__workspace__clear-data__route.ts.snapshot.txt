import { NextRequest, NextResponse } from "next/server";
import { prisma } from "@/lib/prisma";
import { validateUser } from "@/lib/auth";
import { assertWorkspaceWritable } from "@/lib/workspace-readonly-guard";

// 清空工作空间数据
export async function POST(req: NextRequest) {
  try {
    // 验证用户身份
    const authHeader = req.headers.get("authorization");
    const authResult = await validateUser(authHeader);
    
    if (!authResult.valid) {
      return NextResponse.json(
        { error: authResult.error || "UNAUTHORIZED" },
        { status: 401 }
      );
    }

    const userId = authResult.user!.id;
    const { workspaceId, confirmText } = await req.json();

    // 验证确认文本
    if (confirmText !== "确认删除" && confirmText !== "确认重置") {
      return NextResponse.json(
        { error: "确认文本不正确" },
        { status: 400 }
      );
    }

    // 验证工作空间 ID
    if (!workspaceId) {
      return NextResponse.json(
        { error: "缺少工作空间 ID" },
        { status: 400 }
      );
    }

    // 归档 / 停用空间禁止写入（硬只读）
    await assertWorkspaceWritable(workspaceId);

    // 获取工作空间信息
    const workspace = await prisma.workspace.findUnique({
      where: { id: workspaceId },
    });

    if (!workspace) {
      return NextResponse.json(
        { error: "工作空间不存在" },
        { status: 404 }
      );
    }

    // 验证用户是否有权限管理该工作空间 (Owner 拥有绝对权限，或者在 workspacemember 中是 OWNER/ADMIN)
    const isOwner = workspace.ownerId === userId;
    let isAuthorized = isOwner;

    if (!isAuthorized) {
      const membership = await prisma.workspacemember.findFirst({
        where: {
          userId,
          workspaceId,
          role: {
            in: ["OWNER", "ADMIN"],
          },
        },
      });
      if (membership) {
        isAuthorized = true;
      }
    }

    if (!isAuthorized) {
      return NextResponse.json(
        { error: "无权管理该工作空间" },
        { status: 403 }
      );
    }

    // 执行真正的空间数据物理清空重置 (彻底清空该空间下所有历史任务、组件装配、文档资料与知识规约)
    // 1. 删除组件任务运行历史（任务按 tenantId 关联工作空间）
    await prisma.componenttask.deleteMany({
      where: { tenantId: workspaceId },
    });

    // 2. 删除组件装配绑定与装配历史记录
    await prisma.componentusage.deleteMany({
      where: { workspaceId },
    });

    // 3. 删除归档文档资料与分析报告（含 knowledge 类型的知识库条目）
    //    [R0-J2 J-CLEAR] 安全认领协议：
    //    - 只删除「物理文件已确认删除」且「条件删行认领成功」的对象记录（不再全空间 deleteMany）；
    //    - 文件删除失败/共享引用未清的对象：保留记录、文件关联与计量，如实计入部分失败；
    //    - 扫描开始后新上传的对象不在扫描集合中，文件与记录完整保留；
    //    - 与另一删除/清空操作竞争：条件删行（id+workspaceId）只有一方 count=1，占用只释放一次；
    //    - 释放漂移（used < bytes）：不扣量，登记孤儿恢复线索，由对账呈现差异。
    const removingDocs = await prisma.document.findMany({
      where: { workspaceId },
      select: { id: true, filePath: true, fileSize: true, title: true },
    });
    const { deleteAssetFile } = await import("@/lib/file-store");
    const { releaseStorageBytes, persistOrphanCleanupLead } = await import("@/lib/storage-quota");
    let clearedCount = 0;
    const failedObjects: Array<{ id: string; title: string | null; reason: string }> = [];
    let sharedRefKept = 0;
    let concurrentSkipped = 0;
    let driftSkipped = 0;
    for (const d of removingDocs) {
      // 共享引用保护：同 filePath 仍有其他在库文档引用 → 保留文件与记录，不释放占用
      if (d.filePath) {
        const stillReferenced = await prisma.document.count({
          where: { workspaceId, filePath: d.filePath, id: { not: d.id } },
        });
        if (stillReferenced > 0) {
          sharedRefKept++;
          continue;
        }
      }
      // 物理删除先行并核验：失败保留该对象的记录与计量（不释放）
      const fileDel = await deleteAssetFile(d.filePath);
      if (!fileDel.ok) {
        console.error(`[clear-data] 物理文件删除失败，保留记录与计量: ${d.id}`, fileDel.error);
        failedObjects.push({ id: d.id, title: d.title ?? null, reason: "物理文件删除失败（" + (fileDel.error || "未知原因") + "）" });
        continue;
      }
      // 条件删行认领：count=1 才由本次清空负责释放；count=0 = 并发删除已处理（对方协议已释放），不重复扣量
      const claimed = await prisma.document.deleteMany({ where: { id: d.id, workspaceId } });
      if (claimed.count !== 1) {
        concurrentSkipped++;
        continue;
      }
      const rel = await releaseStorageBytes(workspaceId, d.fileSize ?? 0);
      if (rel.drift) {
        // 漂移：文件已删但占用不足以扣减 → 登记恢复线索，不盲目清零
        driftSkipped++;
        await persistOrphanCleanupLead(workspaceId, d.filePath ?? null, d.fileSize ?? null, "clear-data: 文件已删除但释放漂移，占用未扣减");
      }
      clearedCount++;
    }

    // 4. 删除该空间的历史操作审计日志 (确保空间彻底还原为物理出厂初始状态)
    await prisma.operationlog.deleteMany({
      where: { workspaceId },
    });

    // 5. 更新工作空间的 updatedAt 时间戳为最新的重置时间 (独立 try-catch 保证绝对写入成功)
    try {
      const resetTime = new Date();
      await prisma.workspace.update({
        where: { id: workspaceId },
        data: { updatedAt: resetTime },
      });
      console.log(`[clear-data] 工作空间 ${workspaceId} updatedAt 成功更新为: ${resetTime.toISOString()}`);
    } catch (updateErr) {
      console.error(`[clear-data] 更新工作空间 ${workspaceId} updatedAt 失败:`, updateErr);
    }

    // 6. 对个人空间自动进行出厂默认组件装配恢复 (百分百重新自动装配 5 个初始化默认通用效能组件)
    if (workspace.type === "PERSONAL") {
      const { ensureDefaultComponents } = await import("@/lib/workspaceInit");
      await ensureDefaultComponents(workspaceId, userId);
    }

    // P3：空间数据已清空，立即失效组件数/中枢等只读缓存，保证回中枢后计数即时归位
    const { clearServerCache } = await import("@/lib/serverCache");
    clearServerCache();

    // [R0-J2 J-CLEAR] 部分失败如实呈现：失败对象保留文件与记录，不假报全部完成
    const hasPartialFailure = failedObjects.length > 0 || driftSkipped > 0;
    const partialNote = hasPartialFailure
      ? `其中 ${failedObjects.length} 个对象的物理文件删除失败（已保留记录与计量）、${driftSkipped} 个对象释放漂移已登记待核实${sharedRefKept > 0 ? `、${sharedRefKept} 个对象因共享引用保留` : ""}。`
      : "";

    return NextResponse.json({
      success: true,
      message: hasPartialFailure
        ? `空间已重置：${clearedCount} 个资料对象已彻底清除，${failedObjects.length} 个失败对象已保留可重试依据。${partialNote}`
        : "个人空间已成功物理重置为纯净出厂状态，5 个出厂默认组件已重新装配",
      resetAt: new Date().toISOString(),
      storage: {
        clearedCount,
        failedCount: failedObjects.length,
        failedObjects,
        sharedRefKept,
        concurrentSkipped,
        driftSkipped,
      },
    });
  } catch (error) {
    console.error("Clear workspace data error:", error);
    return NextResponse.json(
      { error: "清空工作空间数据失败" },
      { status: 500 }
    );
  }
}
